1. Wait for it to be ready:

   kubectl rollout status {{ if eq .Values.engine "ferretdb" }}statefulset/{{ include "mongo-cloud-pod.postgresName" . }}{{ else }}statefulset/{{ include "mongo-cloud-pod.mongodbName" . }}{{ end }} -n {{ .Release.Namespace }}
{{- if eq .Values.engine "ferretdb" }}
   kubectl rollout status deployment/{{ include "mongo-cloud-pod.gatewayName" . }} -n {{ .Release.Namespace }}
{{- end }}

2. Get the connection string:
{{- if .Values.auth.existingSecret }}

   Credentials come from your own Secret "{{ .Values.auth.existingSecret }}".
   Host: {{ include "mongo-cloud-pod.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.port }}
{{- else }}

   kubectl get secret {{ include "mongo-cloud-pod.fullname" . }} -n {{ .Release.Namespace }} \
     -o jsonpath='{.data.connectionString}' | base64 -d; echo

   In-cluster hostname:
     {{ include "mongo-cloud-pod.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.port }}
{{- end }}

3. Verify it end to end (insert, index, aggregate, delete):

   helm test {{ .Release.Name }} -n {{ .Release.Namespace }}

4. Connect from your laptop:

   kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "mongo-cloud-pod.fullname" . }} 27017:{{ .Values.service.port }}
   mongosh "mongodb://{{ .Values.auth.username }}:<password>@localhost:27017/{{ .Values.auth.database }}{{ if eq .Values.engine "mongodb-community" }}?authSource=admin{{ end }}"
